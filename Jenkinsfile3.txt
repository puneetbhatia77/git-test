pipeline {
    agent any
    stages {
        stage('Test') {
            steps {
                // simulate tests failing 
                     catchError(buildResult: 'UNSTABLE', stageResult: 'UNSTABLE') {
                     echo 'Fail!'; sh 'exit 1'          
                // echo 'SUCCESS'
            }
         }
        }
        stage('Mark explicitly if needed') {
              steps {
                script {
                  if (currentBuild.result == 'UNSTABLE') {
                    echo "Tests failed, marking build unstable"
                    currentBuild.result = 'UNSTABLE'
                  }
            }
          }
        }
    }
    post {
        always {
            echo 'always : This will always run'
        }
        success {
            echo 'success : This will run only if successful'
        }
        failure {
            echo 'failure : This will run only if failed'
        }
        unstable {
            echo 'unstable : This will run only if the run was marked as unstable'
        }
        changed {
            echo 'changed : This will run only if the state of the Pipeline has changed from success to fail and vice-versa'
           
        }
    }
}
